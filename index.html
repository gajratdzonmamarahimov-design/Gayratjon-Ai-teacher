Ин сайт пурра static аст. Барои 24/7 дар Render - як Node server хурд кифоя аст.

server.js
import express from 'express';
import path from 'path';
const app = express();
app.use(express.static('dist'));
app.get('*',(req,res)=>res.sendFile(path.join(process.cwd(),'dist','index.html')));
app.listen(process.env.PORT||3000,()=>console.log('AI Muallim 24/7'));
Dockerfile
FROM node:20-alpine
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build
EXPOSE 3000
CMD ["node","server.js"]
Build: npm run build
Start: node server.js
